        // OGN APRS パケット解析関数
        function parseOgnPacket(line) {
            try {
                if (!line || line.startsWith('#')) return null;
                const match = line.match(/^([A-Za-z0-9-]+)>[^:]+:\/(\d{6})h(\d{4}\.\d{2})([NS])[\/\\](\d{5}\.\d{2})([EW])([0-9\/>]{3})\/(\d{3})\/A=(\d{6})/);
                if (!match) return null;

                const callsign = match[1];
                const latDeg = parseFloat(match[3].substring(0, 2));
                const latMin = parseFloat(match[3].substring(2));
                let lat = latDeg + (latMin / 60);
                if (match[4] === 'S') lat = -lat;

                const lngDeg = parseFloat(match[5].substring(0, 3));
                const lngMin = parseFloat(match[5].substring(3));
                let lng = lngDeg + (lngMin / 60);
                if (match[6] === 'W') lng = -lng;

                const hdg = parseInt(match[7], 10) || 0;
                const spdKnots = parseInt(match[8], 10) || 0;
                const spdKm = spdKnots * 1.852;

                const altFeet = parseInt(match[9], 10) || 0;
                const altMeters = altFeet * 0.3048;

                let vs = 0;
                const vsMatch = line.match(/([+-]\d+fpm|fpm[+-]\d+)/);
                if (vsMatch) {
                    const fpm = parseInt(vsMatch[1].replace(/[^0-9+-]/g, ''), 10);
                    vs = (fpm * 0.3048) / 60;
                }

                const isGlider = !callsign.includes('TOW') && !callsign.includes('POWER');

                return {
                    icao24: callsign.toLowerCase(),
                    callsign: callsign,
                    isGlider: isGlider,
                    lat: lat,
                    lng: lng,
                    alt: altMeters,
                    spd: spdKm,
                    hdg: hdg,
                    vs: vs,
                    lastSeen: Date.now()
                };
            } catch (e) {
                return null;
            }
        }

        // 国土地理院標高API経由で現在地の地表標高を取得
        async function getGroundElevation(lat, lng) {
            const key = `${lat.toFixed(3)}_${lng.toFixed(3)}`;
            if (groundElevCache[key] !== undefined) {
                return groundElevCache[key];
            }
            try {
                const res = await fetch(`https://cyberjapandata2.gsi.go.jp/general/dem/scripts/getelevation.php?lon=${lng}&lat=${lat}&outtype=JSON`);
                const data = await res.json();
                if (data && typeof data.elevation === 'number') {
                    groundElevCache[key] = data.elevation;
                    return data.elevation;
                }
            } catch (e) {
                // エラー時のフォールバック
            }
            return 35.0; // デフォルト 35m
        }

        // 全機体のフライト追跡 & 動的着陸判定 (+5m)
        async function updateFlightTracksAndLogs(ac) {
            if (isPlaybackMode) return;

            if (!flightTracks[ac.icao24]) {
                flightTracks[ac.icao24] = {
                    icao24: ac.icao24,
                    callsign: ac.callsign,
                    isGlider: ac.isGlider,
                    points: [],
                    startTime: new Date(),
                    inAir: false
                };
            }

            const track = flightTracks[ac.icao24];
            const pt = {
                lat: ac.lat,
                lng: ac.lng,
                alt: ac.alt,
                spd: ac.spd,
                hdg: ac.hdg,
                vs: ac.vs,
                time: new Date()
            };

            // 現在地の標高を取得して離着陸判定閾値を設定
            const groundAlt = await getGroundElevation(ac.lat, ac.lng);
            const landingThresholdAlt = groundAlt + 5.0; // 地表標高 + 5m
            const takeoffThresholdAlt = groundAlt + 15.0; // 地表標高 + 15m

            // 離陸検出
            if (!track.inAir && ac.alt > takeoffThresholdAlt) {
                track.inAir = true;
                track.startTime = new Date();
                track.points = [];
            }

            // ポイント追加
            if (track.inAir || track.points.length === 0) {
                const lastPt = track.points[track.points.length - 1];
                if (!lastPt || (Math.abs(lastPt.lat - pt.lat) > 0.0001 || Math.abs(lastPt.lng - pt.lng) > 0.0001)) {
                    track.points.push(pt);
                }
            }

            // 着陸検出（現在地標高 + 5m 以下）
            if (track.inAir && ac.alt <= landingThresholdAlt && track.points.length > 5) {
                track.inAir = false;
                const endTime = new Date();
                
                const logEntry = {
                    id: ac.icao24 + '_' + track.startTime.getTime(),
                    icao24: ac.icao24,
                    callsign: ac.callsign,
                    isGlider: ac.isGlider,
                    startTime: track.startTime,
                    endTime: endTime,
                    durationMin: Math.max(1, Math.round((endTime - track.startTime) / 60000)),
                    maxAlt: Math.round(Math.max(...track.points.map(p => p.alt))),
                    points: [...track.points]
                };

                if (!flightLogs.some(l => l.id === logEntry.id)) {
                    flightLogs.unshift(logEntry);
                }

                track.points = [];
            }

            if (selectedIcao24 === ac.icao24) {
                drawAircraftTrack(ac.icao24);
            }
        }

        // 軌跡（全て実線）の描画
        function drawAircraftTrack(icao24) {
            clearTrackLines();

            const track = flightTracks[icao24];
            if (!track || !track.points || track.points.length < 2) return;

            const latLngs = track.points.map(p => [p.lat, p.lng]);
            const isGlider = track.isGlider;

            trackPolylines[icao24] = L.polyline(latLngs, {
                color: isGlider ? '#38bdf8' : '#cbd5e1',
                weight: 4,
                opacity: 0.8
            }).addTo(map);
        }

        function clearTrackLines() {
            Object.keys(trackPolylines).forEach(icao => {
                if (trackPolylines[icao]) {
                    map.removeLayer(trackPolylines[icao]);
                    delete trackPolylines[icao];
                }
            });
        }

        // 毎日深夜0時（00:00:00）にデータを自動クリア・リセットする機能
        function setupMidnightReset() {
            const now = new Date();
            const midnight = new Date(now);
            midnight.setHours(24, 0, 0, 0); // 今日の深夜0時（＝明日の開始）

            const timeUntilMidnight = midnight.getTime() - now.getTime();

            setTimeout(() => {
                // データのリセット実行
                flightLogs = [];
                Object.keys(flightTracks).forEach(icao => {
                    if (flightTracks[icao]) {
                        flightTracks[icao].points = [];
                    }
                });
                
                clearTrackLines();
                updateUI();

                console.log("【自動リセット】深夜0時を通過したため、本日のフライトログと軌跡履歴を消去しました。");
                
                // 次の日の0時用に再セット
                setupMidnightReset();
            }, timeUntilMidnight);
        }

        // OGN WebSocket 接続
        function connectOGN() {
            const ognWsUrl = 'wss://aprs.glidernet.org:30012/';
            document.getElementById('system-status').innerText = "OGN 接続中...";
            document.getElementById('system-status').className = "text-yellow-400";

            try {
                ognSocket = new WebSocket(ognWsUrl);

                ognSocket.onopen = () => {
                    isUsingOgn = true;
                    document.getElementById('data-source').innerText = "OGN (Live WebSocket)";
                    document.getElementById('system-status').innerText = "OGN Live 接続完了";
                    document.getElementById('system-status').className = "text-emerald-400";
                    
                    const filterCmd = `user MenumaWebApp pass -1 filter r/${MENUMA_LAT.toFixed(2)}/${MENUMA_LNG.toFixed(2)}/50\r\n`;
                    ognSocket.send(filterCmd);
                };

                ognSocket.onmessage = (event) => {
                    if (isPlaybackMode) return;
                    const parsed = parseOgnPacket(event.data);
                    if (parsed) {
                        const idx = currentData.findIndex(d => d.icao24 === parsed.icao24);
                        if (idx >= 0) {
                            currentData[idx] = { ...currentData[idx], ...parsed };
                        } else {
                            currentData.push(parsed);
                        }
                        updateFlightTracksAndLogs(parsed);
                        updateUI();
                    }
                };

                ognSocket.onerror = (err) => {
                    document.getElementById('system-status').innerText = "OGN 接続エラー";
                    document.getElementById('system-status').className = "text-red-400";
                };

                ognSocket.onclose = () => {
                    document.getElementById('system-status').innerText = "OGN 切断";
                    document.getElementById('system-status').className = "text-rose-400";
                };

            } catch (e) {
                document.getElementById('system-status').innerText = "OGN 接続不可";
                document.getElementById('system-status').className = "text-rose-400";
            }
        }

        // プレイバック機能ロジック
        function startPlayback(log) {
            isPlaybackMode = true;
            activePlaybackLog = log;
            playbackCurrentIndex = 0;
            isPlaybackPlaying = false;

            document.getElementById('flight-details').classList.add('hidden');
            document.getElementById('playback-bar').classList.remove('hidden');
            
            document.getElementById('pb-callsign').innerText = log.callsign;
            const startStr = log.startTime.toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            const endStr = log.endTime.toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            document.getElementById('pb-time-range').innerText = `${startStr} - ${endStr}`;
            
            const slider = document.getElementById('pb-slider');
            slider.max = log.points.length - 1;
            slider.value = 0;

            Object.values(aircraftMarkers).forEach(m => map.removeLayer(m));
            clearTrackLines();

            const latLngs = log.points.map(p => [p.lat, p.lng]);
            if (playbackPolyline) map.removeLayer(playbackPolyline);
            playbackPolyline = L.polyline(latLngs, { color: '#38bdf8', weight: 4, opacity: 0.8 }).addTo(map);

            map.fitBounds(playbackPolyline.getBounds(), { padding: [50, 50] });

            updatePlaybackFrame();
            togglePlaybackPlay(true);
        }

        function updatePlaybackFrame() {
            if (!activePlaybackLog || !activePlaybackLog.points.length) return;

            const pt = activePlaybackLog.points[playbackCurrentIndex];
            const slider = document.getElementById('pb-slider');
            slider.value = playbackCurrentIndex;

            const timeStr = pt.time.toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
            document.getElementById('pb-current-time').innerText = timeStr;

            if (playbackMarker) map.removeLayer(playbackMarker);
            
            let alertLevel = 0;
            if (pt.alt > 35 && pt.vs < 0) {
                if (pt.alt < 200) alertLevel = 3;
                else if (pt.alt < 280) alertLevel = 2;
                else if (pt.alt < 300) alertLevel = 1;
            }

            playbackMarker = L.marker([pt.lat, pt.lng], {
                icon: getPlaneIcon(pt.hdg, activePlaybackLog.isGlider, alertLevel)
            }).addTo(map);

            const detailsPanel = document.getElementById('flight-details');
            detailsPanel.classList.remove('hidden');
            
            document.getElementById('detail-callsign').innerText = activePlaybackLog.callsign;
            document.getElementById('detail-model').innerText = activePlaybackLog.isGlider ? 'Glider (Playback)' : 'Tow Plane (Playback)';
            document.getElementById('detail-alt').innerText = Math.round(pt.alt);
            document.getElementById('detail-spd').innerText = Math.round(pt.spd);
            document.getElementById('detail-hdg').innerText = Math.round(pt.hdg);
            
            const vsEl = document.getElementById('detail-vs');
            vsEl.innerText = (pt.vs > 0 ? '+' : '') + pt.vs.toFixed(1);
            document.getElementById('detail-lat').innerText = pt.lat.toFixed(4);
            document.getElementById('detail-lng').innerText = pt.lng.toFixed(4);
        }

        function togglePlaybackPlay(forcePlay) {
            if (forcePlay !== undefined) isPlaybackPlaying = forcePlay;
            else isPlaybackPlaying = !isPlaybackPlaying;

            const icon = document.getElementById('pb-play-icon');
            if (isPlaybackPlaying) {
                icon.innerHTML = '<path d="M6 19h4V5H6v14zm8-14v14h4V5h-4z"/>';
                runPlaybackLoop();
            } else {
                icon.innerHTML = '<path d="M8 5v14l11-7z"/>';
                clearTimeout(playbackTimer);
            }
        }

        function runPlaybackLoop() {
            clearTimeout(playbackTimer);
            if (!isPlaybackPlaying || !activePlaybackLog) return;

            if (playbackCurrentIndex < activePlaybackLog.points.length - 1) {
                playbackCurrentIndex++;
                updatePlaybackFrame();
                playbackTimer = setTimeout(runPlaybackLoop, 1000 / playbackSpeed);
            } else {
                togglePlaybackPlay(false);
            }
        }

        function onPlaybackSeek(val) {
            playbackCurrentIndex = parseInt(val, 10);
            updatePlaybackFrame();
        }

        function changePlaybackSpeed(speed) {
            playbackSpeed = parseFloat(speed);
        }

        function exitPlayback() {
            isPlaybackMode = false;
            togglePlaybackPlay(false);

            if (playbackMarker) map.removeLayer(playbackMarker);
            if (playbackPolyline) map.removeLayer(playbackPolyline);
            
            document.getElementById('playback-bar').classList.add('hidden');
            document.getElementById('flight-details').classList.add('hidden');

            updateUI();
        }

        // マップ初期化
        function initMap() {
            const googleRoadmaps = L.tileLayer('https://{s}.google.com/vt/lyrs=m&x={x}&y={y}&z={z}', {
                maxZoom: 20, subdomains: ['mt0', 'mt1', 'mt2', 'mt3'], attribution: '&copy; Google Maps'
            });

            const googleHybrid = L.tileLayer('https://{s}.google.com/vt/lyrs=y&x={x}&y={y}&z={z}', {
                maxZoom: 20, subdomains: ['mt0', 'mt1', 'mt2', 'mt3'], attribution: '&copy; Google Maps'
            });

            const googleTerrain = L.tileLayer('https://{s}.google.com/vt/lyrs=p&x={x}&y={y}&z={z}', {
                maxZoom: 20, subdomains: ['mt0', 'mt1', 'mt2', 'mt3'], attribution: '&copy; Google Maps'
            });

            const osm = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                attribution: '&copy; OpenStreetMap contributors', maxZoom: 19
            });

            map = L.map('map', {
                center: [MENUMA_LAT, MENUMA_LNG],
                zoom: 15,
                layers: [googleRoadmaps]
            });

            const baseMaps = {
                "Google Maps (標準)": googleRoadmaps,
                "Google Maps (航空写真)": googleHybrid,
                "Google Maps (地形)": googleTerrain,
                "OpenStreetMap": osm
            };
            L.control.layers(baseMaps).addTo(map);

            L.marker([MENUMA_LAT, MENUMA_LNG]).addTo(map).bindPopup('<b>妻沼滑空場</b><br>Menuma Gliderport');

            map.on('moveend', updateUI);
        }

        function selectAircraft(icao24) {
            selectedIcao24 = icao24;
            drawAircraftTrack(icao24);
            updateUI();
        }

        function closeDetails() {
            selectedIcao24 = null;
            clearTrackLines();
            updateUI();
        }

        // UI・マップマーカー・機体詳細リストの更新
        function updateUI() {
            if (isPlaybackMode) return;

            const sidebarArea = document.getElementById('sidebar-content-area');
            sidebarArea.innerHTML = '';

            const bounds = map ? map.getBounds() : null;
            const visibleData = bounds 
                ? currentData.filter(ac => bounds.contains([ac.lat, ac.lng]))
                : currentData;

            document.getElementById('total-aircrafts').innerText = visibleData.length;
            
            const now = new Date();
            document.getElementById('last-update').innerText = now.toLocaleTimeString('ja-JP');

            const cutoff = Date.now() - 30000;
            currentData = currentData.filter(ac => !ac.lastSeen || ac.lastSeen > cutoff);

            // 全マーカーの更新
            currentData.forEach(ac => {
                let alertLevel = 0;
                if (ac.alt > 35 && ac.vs < 0) {
                    if (ac.alt < 200) alertLevel = 3;
                    else if (ac.alt < 280) alertLevel = 2;
                    else if (ac.alt < 300) alertLevel = 1;
                }

                if (aircraftMarkers[ac.icao24]) {
                    aircraftMarkers[ac.icao24].setLatLng([ac.lat, ac.lng]);
                    aircraftMarkers[ac.icao24].setIcon(getPlaneIcon(ac.hdg, ac.isGlider, alertLevel));
                } else {
                    const marker = L.marker([ac.lat, ac.lng], { icon: getPlaneIcon(ac.hdg, ac.isGlider, alertLevel) }).addTo(map);
                    marker.on('click', () => selectAircraft(ac.icao24));
                    aircraftMarkers[ac.icao24] = marker;
                }
            });

            // マーカー消去
            Object.keys(aircraftMarkers).forEach(icao => {
                if (!currentData.find(d => d.icao24 === icao)) {
                    map.removeLayer(aircraftMarkers[icao]);
                    delete aircraftMarkers[icao];
                }
            });

            // 機体が未選択の場合はメイン機体リストを表示
            if (!selectedIcao24) {
                if (visibleData.length === 0) {
                    sidebarArea.innerHTML = '<div class="flex items-center justify-center h-full text-slate-500 text-sm">画面内に機体はいません</div>';
                } else {
                    visibleData.forEach(ac => {
                        let alertLevel = 0;
                        if (ac.alt > 35 && ac.vs < 0) {
                            if (ac.alt < 200) alertLevel = 3;
                            else if (ac.alt < 280) alertLevel = 2;
                            else if (ac.alt < 300) alertLevel = 1;
                        }

                        const listItem = document.createElement('div');
                        let borderClass = 'border-slate-700';
                        let textClass = 'text-slate-100';
                        let alertInfoHtml = '';

                        if (alertLevel === 3) {
                            borderClass = 'border-red-500/80';
                            textClass = 'text-red-500 animate-pulse';
                            alertInfoHtml = `<div class="text-[10px] text-red-400 mt-1 font-bold">📍 警告(赤 200m未満): ${ac.lat.toFixed(4)}, ${ac.lng.toFixed(4)}</div>`;
                        } else if (alertLevel === 2) {
                            borderClass = 'border-orange-500/80';
                            textClass = 'text-orange-500 font-bold';
                            alertInfoHtml = `<div class="text-[10px] text-orange-400 mt-1 font-bold">📍 注意(橙 280m未満): ${ac.lat.toFixed(4)}, ${ac.lng.toFixed(4)}</div>`;
                        } else if (alertLevel === 1) {
                            borderClass = 'border-yellow-500/80';
                            textClass = 'text-yellow-400 font-bold';
                            alertInfoHtml = `<div class="text-[10px] text-yellow-400 mt-1 font-bold">📍 留意(黄 300m未満): ${ac.lat.toFixed(4)}, ${ac.lng.toFixed(4)}</div>`;
                        }

                        listItem.className = `p-3 mb-2 rounded-lg cursor-pointer transition-colors border ${borderClass} bg-slate-900 hover:bg-slate-800 flex items-center justify-between`;
                        listItem.onclick = () => {
                            selectAircraft(ac.icao24);
                            map.panTo([ac.lat, ac.lng]);
                        };
                        
                        listItem.innerHTML = `
                            <div>
                                <div class="font-bold flex items-center gap-2 ${textClass}">
                                    ${ac.callsign}
                                    ${ac.isGlider ? '<span class="px-1.5 py-0.5 rounded bg-sky-500/20 text-sky-400 text-[10px] border border-sky-500/30">Glider</span>' : '<span class="px-1.5 py-0.5 rounded bg-slate-500/20 text-slate-300 text-[10px] border border-slate-500/30">Tow</span>'}
                                </div>
                                <div class="text-xs font-mono text-slate-400 mt-1">Alt: ${Math.round(ac.alt)} m | Spd: ${Math.round(ac.spd)} km/h</div>
                                ${alertInfoHtml}
                            </div>
                            <div class="text-slate-500">
                                <svg class="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 5l7 7-7 7"></path></svg>
                            </div>
                        `;
                        sidebarArea.appendChild(listItem);
                    });
                }
            } else {
                // 特定機体が選択された場合：対象機体の過去フライト履歴リストを表示
                const selectedAc = currentData.find(d => d.icao24 === selectedIcao24);
                const callsignName = selectedAc ? selectedAc.callsign : selectedIcao24.toUpperCase();

                const backHeader = document.createElement('div');
                backHeader.className = 'mb-3 pb-2 border-b border-slate-800 flex items-center justify-between';
                backHeader.innerHTML = `
                    <button onclick="closeDetails()" class="text-xs text-sky-400 hover:underline flex items-center gap-1 font-semibold">
                        ← 画面内機体一覧に戻る
                    </button>
                    <span class="font-mono font-bold text-sm text-white">${callsignName} のフライト履歴</span>
                `;
                sidebarArea.appendChild(backHeader);

                const acLogs = flightLogs.filter(l => l.icao24 === selectedIcao24);

                if (acLogs.length === 0) {
                    const emptyEl = document.createElement('div');
                    emptyEl.className = 'text-center py-8 text-slate-500 text-xs';
                    emptyEl.innerText = '本日完了したフライトログはまだありません。';
                    sidebarArea.appendChild(emptyEl);
                } else {
                    acLogs.forEach(log => {
                        const startStr = log.startTime.toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit' });
                        const endStr = log.endTime.toLocaleTimeString('ja-JP', { hour: '2-digit', minute: '2-digit' });

                        const item = document.createElement('div');
                        item.className = 'p-3 mb-2 bg-slate-800 hover:bg-slate-700/80 border border-slate-700 hover:border-sky-500/50 rounded-lg cursor-pointer transition-all';
                        item.onclick = () => startPlayback(log);

                        item.innerHTML = `
                            <div class="flex items-center justify-between text-xs font-mono">
                                <span class="font-bold text-sky-400">発航〜着陸: ${startStr} - ${endStr}</span>
                                <span class="text-emerald-400 font-semibold">${log.durationMin} 分間</span>
                            </div>
                            <div class="mt-2 text-xs text-slate-300 flex justify-between font-mono">
                                <span>最高高度: ${log.maxAlt}m</span>
                                <span class="text-sky-400 hover:underline">▶ プレイバック再生</span>
                            </div>
                        `;
                        sidebarArea.appendChild(item);
                    });
                }
            }

            // 右上パネル更新
            if (selectedIcao24) {
                const ac = currentData.find(d => d.icao24 === selectedIcao24);
                if (ac) {
                    let alertLevel = 0;
                    if (ac.alt > 35 && ac.vs < 0) {
                        if (ac.alt < 200) alertLevel = 3;
                        else if (ac.alt < 280) alertLevel = 2;
                        else if (ac.alt < 300) alertLevel = 1;
                    }

                    let textClass = 'text-slate-100';
                    if (alertLevel === 3) textClass = 'text-red-500 animate-pulse';
                    else if (alertLevel === 2) textClass = 'text-orange-500 font-bold';
                    else if (alertLevel === 1) textClass = 'text-yellow-400 font-bold';

                    const detailsPanel = document.getElementById('flight-details');
                    detailsPanel.classList.remove('hidden');
                    
                    const callsignEl = document.getElementById('detail-callsign');
                    callsignEl.innerText = ac.callsign;
                    callsignEl.className = `text-2xl font-bold tracking-wider ${textClass}`;
                    
                    document.getElementById('detail-model').innerText = ac.isGlider ? 'Sailplane / Glider' : 'Tow Plane / Light Aircraft';
                    document.getElementById('detail-alt').innerText = Math.round(ac.alt);
                    document.getElementById('detail-spd').innerText = Math.round(ac.spd);
                    document.getElementById('detail-hdg').innerText = Math.round(ac.hdg);
                    
                    const vsEl = document.getElementById('detail-vs');
                    vsEl.innerText = (ac.vs > 0 ? '+' : '') + ac.vs.toFixed(1);
                    document.getElementById('detail-vs-container').className = `text-lg font-mono font-semibold ${ac.vs > 0 ? 'text-sky-400' : (ac.vs < 0 ? 'text-orange-400' : 'text-slate-200')}`;
                    
                    const latEl = document.getElementById('detail-lat');
                    const lngEl = document.getElementById('detail-lng');
                    
                    latEl.innerText = ac.lat.toFixed(4);
                    lngEl.innerText = ac.lng.toFixed(4);
                    
                    let coordClass = '';
                    if (alertLevel === 3) coordClass = 'text-red-400 font-bold';
                    else if (alertLevel === 2) coordClass = 'text-orange-400 font-bold';
                    else if (alertLevel === 1) coordClass = 'text-yellow-400 font-bold';
                    
                    latEl.className = coordClass;
                    lngEl.className = coordClass;
                }
            }
        }
